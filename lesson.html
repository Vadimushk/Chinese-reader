При клике показываем пиньинь wordsHtml += `${word}`; }); div.innerHTML = `
${line.speaker}:${wordsHtml}
${line.ru}
`; dialogue.appendChild(div); }); // Переключатель для перевода (пиньинь теперь показывается по клику) const transBtn = document.getElementById('toggleTranslation'); transBtn.onclick = () => { transBtn.classList.toggle('active'); document.querySelectorAll('.translation').forEach(el => el.classList.toggle('hidden', !transBtn.classList.contains('active'))); }; // Удаляем кнопку пиньиня, так как он теперь по клику const pinyinBtn = document.getElementById('togglePinyin'); if (pinyinBtn) pinyinBtn.style.display = 'none'; // Навигация const nav = document.getElementById('nav'); if (idx > 0) { nav.innerHTML += `← Урок ${data.lessons[idx-1].num}`; } if (idx < data.lessons.length - 1) { nav.innerHTML += `Урок ${data.lessons[idx+1].num} →`; } });
